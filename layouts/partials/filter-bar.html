{{- $rows := . -}} {{- /* Built at build time, so the facet list is correct with JavaScript disabled and a task added to any card gets its own checkbox with no edit here. */ -}} {{- $tasks := slice -}} {{- $backgrounds := slice -}} {{- $contexts := slice -}} {{- $compositions := slice -}} {{- $years := slice -}} {{- $licenses := slice -}} {{- range $rows -}} {{- /* One checkbox per member, not one per whole task list. */ -}} {{- range .Params.task -}} {{- if not (in $tasks .) }}{{ $tasks = $tasks | append . }}{{ end -}} {{- end -}} {{- range .Params.background -}} {{- if not (in $backgrounds .) }}{{ $backgrounds = $backgrounds | append . }}{{ end -}} {{- end -}} {{- $context := cond (eq .Params.has_contextual_data true) "included" "images-only" -}} {{- if not (in $contexts $context) }}{{ $contexts = $contexts | append $context }}{{ end -}} {{- with .Params.composition }}{{ if not (in $compositions .) }}{{ $compositions = $compositions | append . }}{{ end }}{{ end -}} {{- with .Params.year }}{{ if not (in $years .) }}{{ $years = $years | append . }}{{ end }}{{ end -}} {{- with .Params.license }}{{ if not (in $licenses .) }}{{ $licenses = $licenses | append . }}{{ end }}{{ end -}} {{- end -}} {{- /* Facets whose row attribute is a `;`-separated list. Declared here rather than restated in the JavaScript. */ -}}
Task {{- range sort $tasks }} {{- end }}
Background {{- range sort $backgrounds }} {{- end }}
Contextual data {{- range sort $contexts }} {{- end }}
Composition {{- range sort $compositions }} {{- end }}
Year {{- range sort $years }} {{- end }}
License {{- range sort $licenses }} {{- end }}
